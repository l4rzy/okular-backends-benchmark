cmake_minimum_required(VERSION 3.20)
project(okular-backend-bench LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ECM 6.3 CONFIG REQUIRED)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(Okular6 REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_path(OKULAR_CORE_INCLUDE_DIR okularcore_export.h PATH_SUFFIXES okular/core REQUIRED)

add_executable(okular-bench main.cpp)
target_include_directories(okular-bench PRIVATE ${OKULAR_CORE_INCLUDE_DIR})
target_link_libraries(okular-bench PRIVATE Qt6::Widgets Okular::Core KF6::CoreAddons KF6::ConfigCore)
target_compile_definitions(okular-bench PRIVATE BENCH_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
target_compile_options(okular-bench PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(test_bench_workload test_workload.cpp)
    set_target_properties(test_bench_workload PROPERTIES AUTOMOC ON)
    target_link_libraries(test_bench_workload PRIVATE Qt6::Test)
    add_test(NAME bench_workload COMMAND test_bench_workload)
    add_test(NAME bench_python COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p "test_*.py")
    add_test(NAME bench_help COMMAND okular-bench --help)
    set_tests_properties(bench_help PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic")
endif()
